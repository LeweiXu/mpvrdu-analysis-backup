 $$ c\left(t\right)=\frac{-ab}{\left(a^{2}\sin^{2}\left(t\right)+b^{2}\cos^{2}\left(t\right)\right)^{3/2}} $$

 $$ \left\{\begin{array}{l}x(t)=a\cos(t)\\ y(t)=b\sin(t)\end{array}\right.,a>0,b>0 $$
